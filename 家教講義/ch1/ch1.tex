\lhead{第一章練習題}

\section{基礎題}
\begin{enumerate}
  \item 將$0.1\overline{23}$化為分數 \vspace{2cm}
  \item 已知$a, b$為有理數，若$\sqrt{7+4\sqrt{3}}a+\sqrt{4-2\sqrt{3}}b\;=7-\sqrt{3}$，求$a, b$\vspace{2cm}
  \item 比較$\cfrac{365}{354}, \cfrac{104}{93}, \cfrac{200}{189}$的大小關係\vspace{2cm}
  \item 已知$a, b\in\mathbb{R}$，若$|ax+1|\leq b$之解為$-1\leq x\leq 9$，求$a, b$\vspace{2cm}
  \item 若$\cfrac{-6x+1}{2x^2+x-3}=\cfrac{A}{x-1}+\cfrac{B}{2x+3}$，求$A+B$\vspace{2cm}
  \item 設$f(x)=\cfrac{1}{\sqrt{x}+\sqrt{x-1}}$，求$f(101)+f(102)+f(103)+...+f(200)$\vspace{2cm}
  \item 若方程式$4^x-5\cdot 2^{x+2}+16=0$之解為$a, b$，求$a+b$\vspace{2cm}
  \item 計算$(\cfrac{81}{16})^{-0.25}\cdot(\cfrac{8}{27})^{-\frac{2}{3}}\cdot(0.25)^{-2.5}$\vspace{2cm}
  \item 計算$10^{\log{(\log{25})}+\log{(1+\log{\frac{\log{4}}{\log{25}}})}}$\vspace{2cm}
  \item 化簡$\log{6}+log{12}-2\log{3}$\vspace{2cm}
\end{enumerate}

\section{進階題}
\begin{enumerate}
    \item 比較$\sqrt{7}-\sqrt{2}$與$\sqrt{8}-\sqrt{3}$的大小\vspace{2cm}
    \item 已知$0<x<1$，求$\sqrt{x^2+\frac{1}{x^2}+2}-\sqrt{x^2+\frac{1}{x^2}-2}$（請使用$x$表示）\vspace{2cm}
    \item 若$a+\frac{1}{a}=4$，求$a^5+\cfrac{1}{a^5}$\vspace{2cm}
    \item 求$|x-3|+|2x+1|+4x-5=7$之解\vspace{2cm}
    \item 設$a>0$，若$2xy-4x+y=0$且$a^x=10^y=25$，求$a$\vspace{2cm}
\end{enumerate}

\section{困難題}
\begin{enumerate}
    \item 若$a+b=3, ab=2$，求$a^5+b^5$\vspace{2cm}
    \item 計算$\cfrac{(1993^2-1999)\times (1993^2+3986-3)\times 1994}{1990\times 1992 \times 1995 \times 1996}$\vspace{2cm}
    \item 在算幾不等式中，我們知道若$a, b\in\mathbb{R}^+$，則$\cfrac{a+b}{2}\geq\sqrt{ab}$。已知$a, b>0$且$2a+3b=8$，求$a\cdot b$之最大值\vspace{2cm}
    \item 解$||x-2| -5|=4$\vspace{2cm}
\end{enumerate}
