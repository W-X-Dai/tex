    \begin{frame}
        \frametitle{What is Vector?}
        In mathamatics, a vector is an object that has both magnitude and direction. But in programming, a vector is a dynamic array that can resize itself automatically when elements are added or removed. It provides random access to its elements and is implemented as a contiguous block of memory.

        You can consider vector as a flexible array that can grow or shrink in size as needed. It is a part of the C++ Standard Template Library (STL) and is widely used for storing and manipulating collections of data.
    \end{frame}

    \begin{frame}[fragile]
        \frametitle{Declaration of vector}
        \begin{minted}{cpp}
            // declaration:
            vector<TYPE> name;
            // declaration with size:
            vector<TYPE> name(SIZE);
            // declaration a multidimensional vector:
            vector<TYPE> name[SIZE];
        \end{minted}
        Note that if you declare a vector w/ \texttt{vector<TYPE> name}, it will be initially empty.
    \end{frame}

    \begin{frame}[fragile]
        \frametitle{Usage of vector}
        \begin{minted}{cpp}
            // access elements:
            name[index];
            // add elements:
            name.push_back(value);
            name.emplace_back(value);
            // remove elements:
            name.pop_back();
        \end{minted}
    \end{frame}

    \begin{frame}[fragile]
        \frametitle{Usage of vector}
        \begin{minted}{cpp}
            // access elements:
            name[index];
            // access it's size:
            name.size();
            // remove all elements:
            name.clear();
        \end{minted}
    \end{frame}

    \begin{frame}[fragile]
        \frametitle{Example}
        \begin{minted}{cpp}
            vector<int> v;
            v.push_back(1);
            v.push_back(2);
            v.push_back(3);
            for(int i=0;i<v.size();++i){
                cout<<v[i]<<" ";
           }
        \end{minted}
    \end{frame}
